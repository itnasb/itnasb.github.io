Automating Workflows

Visual Basic for Applications VBA Automation

Summary

This VBA macro automates the review of inventory records to identify entries that may require follow-up or additional tracking. It analyzes fields in spreadsheet data, and applies rule-based checks with the goal of reducing the amount of manual effort required to review large spreadsheets by filtering out entries not requiring action and highlighting those that may.

How It Works

The macro loops through rows in source worksheets and inspects key fields. Based on these values, the script determines whether the row should be hidden, highlighted for manual review, or left visible for follow-up.

Each record ID is also compared against a separate tracking worksheet. If the record already exists and has not changed, the row is hidden. If the record exists but the description has changed, the script updates the tracking sheet and marks the update for visibility. Records that don't yet exist in the tracking list remain visible so they can be added to the review workflow.

Impact

This approach turns a large spreadsheet review task into a much smaller, prioritized list of items that actually require attention. 


' This VBA macro reviews source worksheets to determine whether
' records require follow-up, should be suppressed from review, or need to
' update an existing entry in a tracking worksheet.

Sub EvaluateSourceSheetA()

Dim rowIndex As Long, recordIdMatch As Variant, statusMatch As Variant

' Using the worksheet by name rather than sheet index
With Worksheets("SourceSheetA")

    'loop through rows containing record identifiers
    For rowIndex = 7 To .Cells(.Rows.Count, "A").End(xlUp).Row

        ' If column AB does not contain markerA or markerB, hide the row
        If InStr(1, .Cells(rowIndex, 10).Value2, "MarkerA", vbTextCompare) = 0 _
        And InStr(1, .Cells(rowIndex, 10).Value2, "MarkerB", vbTextCompare) = 0 Then
        
            .Rows(rowIndex).Hidden = True

        ' Else if the classification fields contain exception markers, highlight the row green
        ElseIf InStr(1, .Cells(rowIndex, 5).Value2, "exceptionFlag", vbTextCompare) > 0 _
        Or InStr(1, .Cells(rowIndex, 10), "exceptionMarker1", vbTextCompare) > 0 Then
        
            .Rows(rowIndex).Interior.ColorIndex = 4
        
        End If

        ' Check if record already exists in tracking sheet
        recordIdMatch = Application.Match(.Cells(rowIndex, "A").Value2, _
                            Worksheets("TrackingSheet").Columns("E"), 0)
        ' If a matching identifier is found, compare the description/status
        If Not IsError(recordIdMatch) Then

            'Check whether description/status has changed
            statusMatch = Application.Match(.Cells(rowIndex, "D").Value2, Worksheets("TrackingSheet").Cells(recordIdMatch, "I"), 0)
            ' If the description/status already matches, hide the row
            If Not IsError(statusMatch) Then

                'No change hide row
                .Rows(rowIndex).Hidden = True

            Else
                'Update changed description/status
                Worksheets("TrackingSheet").Cells(recordIdMatch, "I").Value = .Cells(rowIndex, "D").Value
                .Rows(rowIndex).Hidden = True
            End If
        ' If no matching identifier is found, leave the row visible for follow-up
        End If

    Next rowIndex

End With
End Sub

Sub EvaluateSourceSheetB()

'  Hide rows that don't require follow-up based on marker values or exception flags.

Dim rowIndex As Long, recordIdMatch As Variant, statusMatch As Variant

' Using the worksheet by name rather than sheet index
With Worksheets("SourceSheetB")

    For rowIndex = 7 To .Cells(.Rows.Count, "A").End(xlUp).Row

        'Hide rows meeting exclusion criteria
        If InStr(1, .Cells(rowIndex, 10).Value2, "MarkerC", vbTextCompare) > 0 _
        Or InStr(1, .Cells(rowIndex, 5).Value2, "exceptionFlag2", vbTextCompare) > 0 _
        Or InStr(1, .Cells(rowIndex, 10), "excludeMarker1", vbTextCompare) > 0 Then
            .Rows(rowIndex).Hidden = True
        End If

        ' Check if record already exists in tracking sheet
        recordIdMatch = Application.Match(.Cells(rowIndex, "A").Value2, Worksheets("TrackingSheet").Columns("E"), 0)
        ' If a matching identifier is found, compare the description/status
        If Not IsError(recordIdMatch) Then
            statusMatch = Application.Match(.Cells(rowIndex, "D").Value2, Worksheets("TrackingSheet").Cells(recordIdMatch, "I"), 0)
            
            ' If the description/status already matches, hide the row
            If Not IsError(statusMatch) Then
                'Record unchanged
                .Rows(rowIndex).Hidden = True

            Else
                'Record changed — update tracking sheet
                Worksheets("TrackingSheet").Cells(recordIdMatch, "I").Value = .Cells(rowIndex, "D").Value
                Worksheets("TrackingSheet").Cells(recordIdMatch, "I").Interior.ColorIndex = 4
                .Rows(rowIndex).Hidden = True

            End If

        ' If not in tracking sheet and new records contain review markers
        ElseIf InStr(1, .Cells(rowIndex, 10), "reviewFlag", vbTextCompare) > 0 Then
            .Rows(rowIndex).Interior.ColorIndex = 4

        End If
        ' If the identifier is not in tracking and no review marker is present,
        ' leave the row visible for follow-up
    Next rowIndex
End With
End Sub
